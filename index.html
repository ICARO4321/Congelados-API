const API_URL = "https://congelados-api.onrender.com"; // --- FUNÇÃO DE EXCLUIR PRODUTO --- async function deletarProduto(id) { if (confirm("Deseja realmente excluir este produto?")) { try { const resposta = await fetch(`${API_URL}/produtos/${id}`, { method: 'DELETE' }); if (resposta.ok) { alert("Produto excluído com sucesso!"); carregarProdutos(); // Recarrega a tabela de produtos na tela } else { alert("Erro ao excluir o produto."); } } catch (error) { console.error("Erro de conexão:", error); } } } // --- FUNÇÃO DE EXCLUIR CLIENTE --- async function deletarCliente(id) { if (confirm("Deseja realmente excluir este cliente?")) { try { const resposta = await fetch(`${API_URL}/clientes/${id}`, { method: 'DELETE' }); if (resposta.ok) { alert("Cliente excluído com sucesso!"); carregarClientes(); // Recarrega a tabela de clientes na tela } else { alert("Erro ao excluir o cliente."); } } catch (error) { console.error("Erro de conexão:", error); } } } // --- FUNÇÃO DE EXCLUIR VENDA --- async function deletarVenda(id) { if (confirm("Deseja realmente excluir este registro de venda?")) { try { const resposta = await fetch(`${API_URL}/vendas/${id}`, { method: 'DELETE' }); if (resposta.ok) { alert("Venda excluída com sucesso!"); carregarVendas(); // Recarrega a tabela de vendas na tela } else { alert("Erro ao excluir a venda."); } } catch (error) { console.error("Erro de conexão:", error); } } }
